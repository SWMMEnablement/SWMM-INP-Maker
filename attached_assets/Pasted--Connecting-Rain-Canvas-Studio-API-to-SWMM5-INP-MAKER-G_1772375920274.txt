# Connecting Rain Canvas Studio API to SWMM5 INP MAKER

Great idea, Robert. Your Rain Canvas Studio has **66 rainfall patterns** with an API playground already built. Piping that directly into INP MAKER means generated models come with **realistic, design-storm rainfall** instead of placeholder time series. Here's exactly how to do it.

---

## Step 1: Understand What Rain Canvas Provides

Based on the app, the API likely returns:

```json
{
  "pattern": "SCS Type II",
  "depth_total": 2.0,
  "depth_unit": "inches",
  "duration_hours": 6.0,
  "timestep_minutes": 15,
  "cumulative_fractions": [0, 0.01, 0.02, 0.04, 0.08, ...],
  "incremental_depths": [0.02, 0.02, 0.04, 0.08, 0.16, ...],
  "intensities": [0.08, 0.08, 0.16, 0.32, 0.64, ...],
  "metadata": {
    "return_period": "10-yr",
    "region": "Mid-Atlantic",
    "source": "Rain Canvas Studio"
  }
}
```

The key output for SWMM5 is the **incremental depths array** — that becomes the`[TIMESERIES]` section in the .inp file.

```javascript
  // ---- FETCH RAINFALL PATTERN ----

  async getPattern(patternId, params = {}) {
    const {
      depth = 2.0,
      duration = 6.0,
      timestep = 15,
      units = 'inches'
    } = params;

    // Check cache
    const cacheKey = `${patternId}_${depth}_${duration}_${timestep}_${units}`;
    if (this.patternCache.has(cacheKey)) {
      return this.patternCache.get(cacheKey);
    }

    try {
      // Try API endpoint first
      const response = await fetch(`${this.apiBase}/pattern`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          pattern: patternId,
          depth,
          duration_hours: duration,
          timestep_minutes: timestep,
          units
        })
      });

      if (response.ok) {
        const data = await response.json();
        this.patternCache.set(cacheKey, data);
        return data;
      }
    } catch (e) {
      console.warn('Rain Canvas API unavailable, using built-in patterns');
    }

    // Fallback: generate locally using built-in formulas
    const result = this.generatePatternLocally(patternId, depth, duration, timestep, units);
    this.patternCache.set(cacheKey, result);
    return result;
  }

  // ---- LOCAL PATTERN GENERATION (Fallback) ----

  generatePatternLocally(patternId, depth, duration, timestep, units) {
    const numSteps = Math.ceil((duration * 60) / timestep);
    const fractions = this.getDistributionFractions(patternId, numSteps);
    
    // Convert cumulative fractions to incremental depths
    const incrementalDepths = [];
    for (let i = 0; i < fractions.length; i++) {
      const prev = i > 0 ? fractions[i - 1] : 0;
      incrementalDepths.push((fractions[i] - prev) * depth);
    }

    // Build time labels
    const times = [];
    for (let i = 0; i < numSteps; i++) {
      const totalMinutes = i * timestep;
      const hours = Math.floor(totalMinutes / 60);
      const minutes = totalMinutes % 60;
      times.push(`${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}`);
    }

    return {
      pattern: patternId,
      name: this.getPatternName(patternId),
      depth_total: depth,
      depth_unit: units,
      duration_hours: duration,
      timestep_minutes: timestep,
      num_steps: numSteps,
      times,
      cumulative_fractions: fractions,
      incremental_depths: incrementalDepths,
      intensities: incrementalDepths.map(d => d / (timestep / 60)),
      peak_intensity: Math.max(...incrementalDepths.map(d => d / (timestep / 60))),
      peak_step: incrementalDepths.indexOf(Math.max(...incrementalDepths)),
      total_check: incrementalDepths.reduce((a, b) => a + b, 0)
    };
  }

  getDistributionFractions(patternId, numSteps) {
    const t = Array.from({ length: numSteps }, (_, i) => (i + 1) / numSteps);
    
    switch (patternId) {
      case 'scs-type-i':
        return t.map(x => this.scsDistribution(x, 'I'));
      case 'scs-type-ia':
        return t.map(x => this.scsDistribution(x, 'IA'));
      case 'scs-type-ii':
        return t.map(x => this.scsDistribution(x, 'II'));
      case 'scs-type-iii':
        return t.map(x => this.scsDistribution(x, 'III'));
      
      case 'huff-q1':
        return t.map(x => this.huffDistribution(x, 1));
      case 'huff-q2':
        return t.map(x => this.huffDistribution(x, 2));
      case 'huff-q3':
        return t.map(x => this.huffDistribution(x, 3));
      case 'huff-q4':
        return t.map(x => this.huffDistribution(x, 4));
      
      case 'chicago-0.3':
        return this.chicagoDistribution(numSteps, 0.3);
      case 'chicago-0.4':
        return this.chicagoDistribution(numSteps, 0.4);
      case 'chicago-0.5':
        return this.chicagoDistribution(numSteps, 0.5);
      case 'chicago-0.375':
        return this.chicagoDistribution(numSteps, 0.375);
      
      case 'alt-block':
        return this.alternatingBlockDistribution(numSteps, 0.5);
      case 'alt-block-front':
        return this.alternatingBlockDistribution(numSteps, 0.25);
      case 'alt-block-back':
        return this.alternatingBlockDistribution(numSteps, 0.75);
      
      case 'uniform':
        return t.map(x => x);

      default:

    }
  }

  // ---- DISTRIBUTION FORMULAS ----

  scsDistribution(t, type) {
    // Piecewise linear approximation of SCS distributions
    // Based on TR-55 tabulated values
    const tables = {
      'I': [
        [0, 0], [0.083, 0.035], [0.167, 0.076], [0.25, 0.125],
        [0.333, 0.194], [0.417, 0.310], [0.5, 0.515],
        [0.583, 0.624], [0.667, 0.714], [0.75, 0.790],
        [0.833, 0.866], [0.917, 0.936], [1.0, 1.0]
      ],
      'IA': [
        [0, 0], [0.083, 0.050], [0.167, 0.116], [0.25, 0.206],
        [0.333, 0.303], [0.417, 0.515], [0.5, 0.583],
        [0.583, 0.640], [0.667, 0.696], [0.75, 0.753],
        [0.833, 0.826], [0.917, 0.910], [1.0, 1.0]
      ],
      'II': [
        [0, 0], [0.083, 0.011], [0.167, 0.022], [0.25, 0.035],
        [0.333, 0.048], [0.417, 0.072], [0.5, 0.663],
        [0.583, 0.735], [0.667, 0.772], [0.75, 0.820],
        [0.833, 0.868], [0.917, 0.928], [1.0, 1.0]
      ],
      'III': [
        [0, 0], [0.083, 0.010], [0.167, 0.020], [0.25, 0.032],
        [0.333, 0.048], [0.417, 0.072], [0.5, 0.702],
        [0.583, 0.751], [0.667, 0.785], [0.75, 0.830],
        [0.833, 0.876], [0.917, 0.932], [1.0, 1.0]
      ]
    };

    const table = tables[type] || tables['II'];
    
    // Linear interpolation
    for (let i = 1; i < table.length; i++) {
      if (t <= table[i][0]) {
        const t0 = table[i-1][0], t1 = table[i][0];
        const v0 = table[i-1][1], v1 = table[i][1];
        return v0 + (v1 - v0) * (t - t0) / (t1 - t0);
      }
    }
    return 1.0;
  }

  huffDistribution(t, quartile) {
    // Huff distributions using beta function approximation
    const params = {
      1: { a: 1.5, b: 6.0 },   // Front-loaded
      2: { a: 3.0, b: 4.0 },   // Early-middle
      3: { a: 4.0, b: 3.0 },   // Late-middle
      4: { a: 6.0, b: 1.5 }    // Back-loaded
    };
    
    const { a, b } = params[quartile] || params[2];
    return this.incompleteBeta(t, a, b);
  }

  chicagoDistribution(numSteps, r) {
    // Chicago method: creates a peaked hyetograph
    // r = ratio of time to peak / total duration
    const peakStep = Math.floor(r * numSteps);
    const increments = new Array(numSteps).fill(0);
    
    // IDF: i = a / (t + b)^c (generic)
    const a = 1.0, b = 0.1, c = 0.75;
    
    // Before peak (advancing)
    for (let i = peakStep; i >= 0; i--) {
      const tb = (peakStep - i + 0.5) / numSteps;
      increments[i] = a * Math.pow(tb + b, -c) / numSteps;
    }
    
    // After peak (receding)
    for (let i = peakStep + 1; i < numSteps; i++) {
      const tb = (i - peakStep + 0.5) / numSteps;
      increments[i] = a * Math.pow(tb + b, -c) / numSteps;
    }
    
    // Normalize to cumulative fractions
    const total = increments.reduce((s, v) => s + v, 0);
    const normalized = increments.map(v => v / total);
    
    // Convert to cumulative
    const cumulative = [];
    let sum = 0;
    for (const v of normalized) {
      sum += v;
      cumulative.push(sum);
    }
    return cumulative;
  }

  alternatingBlockDistribution(numSteps, peakPosition) {
    // Alternating block method
    const peakStep = Math.floor(peakPosition * numSteps);
    const increments = new Array(numSteps).fill(0);
    
    // Create ranked intensities (highest at center, alternating sides)
    const ranked = Array.from({ length: numSteps }, (_, i) => 1 / (i + 1));
    const total = ranked.reduce((s, v) => s + v, 0);
    const normalizedRanked = ranked.map(v => v / total);
    
    // Place blocks alternating around peak
    let left = peakStep;
    let right = peakStep;
    
    for (let i = 0; i < normalizedRanked.length; i++) {
      if (i === 0) {
        increments[peakStep] = normalizedRanked[i];
      } else if (i % 2 === 1 && right + 1 < numSteps) {
        right++;
        increments[right] = normalizedRanked[i];
      } else if (left - 1 >= 0) {
        left--;
        increments[left] = normalizedRanked[i];
      } else if (right + 1 < numSteps) {
        right++;
        increments[right] = normalizedRanked[i];
      }
    }
    
    // Cumulative
    const cumulative = [];
    let sum = 0;
    for (const v of increments) {
      sum += v;
      cumulative.push(sum);
    }
    return cumulative;
  }

  triangularDistribution(t, peakT) {
    if (t <= peakT) {
      return (t * t) / (2 * peakT);
    } else {
      return 1 - ((1 - t) * (1 - t)) / (2 * (1 - peakT));
    }
  }

  incompleteBeta(x, a, b) {
    // Simple approximation using power function
    // For beta CDF: I_x(a,b)
    if (x <= 0) return 0;
    if (x >= 1) return 1;
    
    // Numerical integration (simple trapezoidal)
    const n = 100;
    let sum = 0;
    const dx = x / n;
    for (let i = 0; i < n; i++) {
      const t = (i + 0.5) * dx;
      sum += Math.pow(t, a - 1) * Math.pow(1 - t, b - 1) * dx;
    }
    
    // Normalize
    let fullSum = 0;
    const dx2 = 1.0 / n;
    for (let i = 0; i < n; i++) {
      const t = (i + 0.5) * dx2;
      fullSum += Math.pow(t, a - 1) * Math.pow(1 - t, b - 1) * dx2;
    }
    
    return sum / fullSum;
  }

  getPatternName(patternId) {
    for (const category of Object.values(this.patternCatalog)) {
      const found = category.find(p => p.id === patternId);
      if (found) return found.name;
    }
    return patternId;
  }

  // ---- GET ALL PATTERNS (for comparison) ----

  getAllPatternIds() {
    const ids = [];
    for (const category of Object.values(this.patternCatalog)) {
      for (const pattern of category) {
        ids.push(pattern.id);
      }
    }
    return ids;
  }

  getCatalog() {
    return this.patternCatalog;
  }
}
```

## Step 3: SWMM INP Rainfall Injector

This is the bridge — takes Rain Canvas output and writes valid SWMM`[RAINGAGES]` and`[TIMESERIES]` sections.

```javascript
// ============================================
// SWMM RAINFALL SECTION GENERATOR
// File: rainfall-injector.js
// Converts Rain Canvas API output to SWMM .inp sections
// ============================================

class RainfallInjector {
  constructor() {
    this.rainCanvas = new RainCanvasClient();
  }

  // ---- MAIN: Generate and inject rainfall into .inp ----

  async injectRainfall(inpContent, rainfallConfig) {
    const {
      patternId = 'scs-type-ii',
      depth = 2.0,
      duration = 6.0,
      timestep = 15,
      units = 'inches',
      startDate = '01/01/2025',
      startTime = '00:00',
      gageName = 'RG1',
      timeseriesName = 'TS_Design_Storm'
    } = rainfallConfig;

    // Fetch pattern from Rain Canvas
    const pattern = await this.rainCanvas.getPattern(patternId, {
      depth, duration, timestep, units
    });

    // Generate SWMM sections
    const timeseriesSection = this.generateTimeseries(
      timeseriesName, pattern, startDate, startTime
    );
    
    const raingageSection = this.generateRaingage(
      gageName, timeseriesName, timestep, units
    );

    // Inject into .inp content
    let modifiedInp = this.replaceOrInsertSection(
      inpContent, '[TIMESERIES]', timeseriesSection
    );
    
    modifiedInp = this.replaceOrInsertSection(
      modifiedInp, '[RAINGAGES]', raingageSection
    );

    // Update subcatchment rain gage assignments
    modifiedInp = this.assignRainGageToSubcatchments(modifiedInp, gageName);

    // Update simulation duration to match storm
    modifiedInp = this.updateSimulationDuration(
      modifiedInp, startDate, startTime, duration
    );

    return {
      inp: modifiedInp,
      pattern,
      sections: {
        timeseries: timeseriesSection,
        raingages: raingageSection
      },
      metadata: {
        patternName: pattern.name || patternId,
        totalDepth: depth,
        duration,
        timestep,
        peakIntensity: pattern.peak_intensity,
        peakTime: pattern.times?.[pattern.peak_step] || 'N/A'
      }
    };
  }

  // ---- GENERATE [TIMESERIES] SECTION ----

  generateTimeseries(name, pattern, startDate, startTime) {
    const lines = [
      `[TIMESERIES]`,
      `;;Name           Date       Time       Value`,
      `;;-------------- ---------- ---------- ----------`
    ];

    // Add comment with pattern info
    lines.push(`;; Generated by Rain Canvas Studio`);
    lines.push(`;; Pattern: ${pattern.name || pattern.pattern}`);
    lines.push(`;; Total Depth: ${pattern.depth_total} ${pattern.depth_unit}`);
    lines.push(`;; Duration: ${pattern.duration_hours} hours`);
    lines.push(`;; Timestep: ${pattern.timestep_minutes} minutes`);
    lines.push(`;; Peak Intensity: ${pattern.peak_intensity?.toFixed(3) || 'N/A'} ${pattern.depth_unit}/hr`);
    lines.push(`;;`);

    // Parse start time
    const [startHour, startMin] = startTime.split(':').map(Number);

    for (let i = 0; i < pattern.incremental_depths.length; i++) {
      const totalMinutes = i * pattern.timestep_minutes;
      const hours = Math.floor(totalMinutes / 60) + startHour;
      const minutes = (totalMinutes % 60) + startMin;
      
      const adjustedHours = hours + Math.floor(minutes / 60);
      const adjustedMinutes = minutes % 60;
      
      const timeStr = `${adjustedHours.toString().padStart(2, '0')}:${adjustedMinutes.toString().padStart(2, '0')}`;
      const value = pattern.incremental_depths[i].toFixed(4);
      
      // SWMM timeseries format: Name  Date  Time  Value
      // Or without date: Name  Time  Value (relative time)
      lines.push(`${name.padEnd(17)}${startDate}  ${timeStr}      ${value}`);
    }

    // Add zero value after storm ends (clean shutoff)
    const endMinutes = pattern.incremental_depths.length * pattern.timestep_minutes;
    const endHours = Math.floor(endMinutes / 60) + startHour;
    const endMins = (endMinutes % 60) + startMin;
    const endTimeStr = `${(endHours + Math.floor(endMins / 60)).toString().padStart(2, '0')}:${(endMins % 60).toString().padStart(2, '0')}`;
    lines.push(`${name.padEnd(17)}${startDate}  ${endTimeStr}      0.0000`);

    return lines.join('\n');
  }

  // ---- GENERATE [RAINGAGES] SECTION ----

  generateRaingage(gageName, timeseriesName, timestep, units) {
    // SWMM format:
    // Name  Format  Interval  SCF  DataSource  SourceName
    //
    // Format: INTENSITY | VOLUME | CUMULATIVE
    // For incremental depths, use VOLUME
    
    const intervalStr = this.formatInterval(timestep);
    const format = 'VOLUME';  // incremental depths
    const scf = '1.0';       // snow catch factor
    
    const lines = [
      `[RAINGAGES]`,
      `;;Name           Format    Interval  SCF       Source`,
      `;;-------------- --------- --------- --------- ----------`,
      `;; Rain Canvas Studio - Design Storm`,
      `${gageName.padEnd(17)}${format.padEnd(10)}${intervalStr.padEnd(10)}${scf.padEnd(10)}TIMESERIES ${timeseriesName}`
    ];

    return lines.join('\n');
  }

  // ---- ASSIGN RAIN GAGE TO ALL SUBCATCHMENTS ----

  assignRainGageToSubcatchments(inpContent, gageName) {
    const lines = inpContent.split('\n');
    let inSubcatchments = false;
    let headerSkipped = false;

    for (let i = 0; i < lines.length; i++) {
      const trimmed = lines[i].trim();
      
      if (trimmed === '[SUBCATCHMENTS]') {
        inSubcatchments = true;
        headerSkipped = false;
        continue;
      }
      if (trimmed.startsWith('[') && trimmed !== '[SUBCATCHMENTS]') {
        inSubcatchments = false;
        continue;
      }

      if (inSubcatchments && !trimmed.startsWith(';') && trimmed.length > 0) {
        const parts = trimmed.split(/\s+/);
        if (parts.length >= 3) {
          // Subcatchment format: Name  RainGage  Outlet  Area  %Imperv  Width  Slope  ...
          parts[1] = gageName;  // Replace rain gage
          lines[i] = parts.join('\t');
        }
      }
    }

    return lines.join('\n');
  }

  // ---- UPDATE SIMULATION DURATION ----

  updateSimulationDuration(inpContent, startDate, startTime, durationHours) {
    const lines = inpContent.split('\n');
    let inOptions = false;
    
    // Calculate end date/time
    // Add 50% extra simulation time for drainage after rain stops
    const totalHours = durationHours * 1.5;
    const extraDays = Math.floor(totalHours / 24);
    const endHours = Math.floor(totalHours % 24);
    const endMinutes = Math.round((totalHours % 1) * 60);
    
    // Simple date addition
    const [month, day, year] = startDate.split('/').map(Number);
    const endDay = day + extraDays;
    const endDate = `${month.toString().padStart(2, '0')}/${endDay.toString().padStart(2, '0')}/${year}`;
    const endTime = `${endHours.toString().padStart(2, '0')}:${endMinutes.toString().padStart(2, '0')}:00`;

    for (let i = 0; i < lines.length; i++) {
      const trimmed = lines[i].trim();
      
      if (trimmed === '[OPTIONS]') { inOptions = true; continue; }
      if (trimmed.startsWith('[') && trimmed !== '[OPTIONS]') { inOptions = false; continue; }
      
      if (inOptions) {
        if (trimmed.startsWith('START_DATE')) {
          lines[i] = `START_DATE           ${startDate}`;
        } else if (trimmed.startsWith('START_TIME')) {
          lines[i] = `START_TIME           ${startTime}:00`;
        } else if (trimmed.startsWith('END_DATE')) {
          lines[i] = `END_DATE             ${endDate}`;
        } else if (trimmed.startsWith('END_TIME')) {
          lines[i] = `END_TIME             ${endTime}`;
        } else if (trimmed.startsWith('REPORT_START_DATE')) {
          lines[i] = `REPORT_START_DATE    ${startDate}`;
        } else if (trimmed.startsWith('REPORT_START_TIME')) {
          lines[i] = `REPORT_START_TIME    ${startTime}:00`;
        }
      }
    }

    return lines.join('\n');
  }

  // ---- INP SECTION MANIPULATION ----

  replaceOrInsertSection(inpContent, sectionHeader, newContent) {
    const lines = inpContent.split('\n');
    const sectionStart = lines.findIndex(l => 
      l.trim().toUpperCase() === sectionHeader.toUpperCase()
    );

    if (sectionStart >= 0) {
      // Find end of existing section
      let sectionEnd = sectionStart + 1;
      while (sectionEnd < lines.length && !lines[sectionEnd].trim().startsWith('[')) {
        sectionEnd++;
      }
      
      // Replace existing section
      lines.splice(sectionStart, sectionEnd - sectionStart, ...newContent.split('\n'));
    } else {
      // Insert new section before [REPORT] or at end
      const reportIdx = lines.findIndex(l => l.trim().toUpperCase() === '[REPORT]');
      const insertIdx = reportIdx >= 0 ? reportIdx : lines.length;
      lines.splice(insertIdx, 0, '', ...newContent.split('\n'), '');
    }

    return lines.join('\n');
  }

  formatInterval(minutes) {
    const hours = Math.floor(minutes / 60);
    const mins = minutes % 60;
    return `${hours}:${mins.toString().padStart(2, '0')}`;
  }

  // ---- MULTI-STORM GENERATION ----

  async injectMultipleStorms(inpContent, storms) {
    // For sensitivity analysis: inject multiple storms as separate time series
    // Each storm gets its own rain gage + timeseries
    
    let modifiedInp = inpContent;
    const allTimeseries = [];
    const allRaingages = [];
    
    for (let i = 0; i < storms.length; i++) {
      const storm = storms[i];
      const gageName = storm.gageName || `RG${i + 1}`;
      const tsName = storm.timeseriesName || `TS_Storm_${i + 1}`;
      
      const pattern = await this.rainCanvas.getPattern(storm.patternId, {
        depth: storm.depth,
        duration: storm.duration,
        timestep: storm.timestep,
        units: storm.units
      });
      
      allTimeseries.push(
        this.generateTimeseries(tsName, pattern, storm.startDate || '01/01/2025', storm.startTime || '00:00')
          .split('\n').filter(l => !l.startsWith('[') && !l.startsWith(';;N') && !l.startsWith(';;-'))
          .join('\n')
      );
      
      allRaingages.push(
        `${gageName.padEnd(17)}VOLUME    ${this.formatInterval(storm.timestep).padEnd(10)}1.0       TIMESERIES ${tsName}`
      );
    }
    
    // Build combined sections
    const combinedTimeseries = [
      `[TIMESERIES]`,
      `;;Name           Date       Time       Value`,
      `;;-------------- ---------- ---------- ----------`,
      `;; Multi-storm design set from Rain Canvas Studio`,
      ...allTimeseries
    ].join('\n');
    
    const combinedRaingages = [
      `[RAINGAGES]`,
      `;;Name           Format    Interval  SCF       Source`,
      `;;-------------- --------- --------- --------- ----------`,
      ...allRaingages
    ].join('\n');
    
    modifiedInp = this.replaceOrInsertSection(modifiedInp, '[TIMESERIES]', combinedTimeseries);
    modifiedInp = this.replaceOrInsertSection(modifiedInp, '[RAINGAGES]', combinedRaingages);
    
    return modifiedInp;
  }
}
```

```javascript
  drawChart(pattern) {
    const canvas = document.getElementById('rain-preview-chart');
    if (!canvas) return;
    
    const ctx = canvas.getContext('2d');
    const dpr = window.devicePixelRatio || 1;
    canvas.width = canvas.offsetWidth * dpr;
    canvas.height = canvas.offsetHeight * dpr;
    ctx.scale(dpr, dpr);
    
    const w = canvas.offsetWidth;
    const h = canvas.offsetHeight;
    const padding = { top: 20, right: 20, bottom: 30, left: 50 };
    const plotW = w - padding.left - padding.right;
    const plotH = h - padding.top - padding.bottom;

    // Clear
    ctx.fillStyle = 'transparent';
    ctx.clearRect(0, 0, w, h);

    const depths = pattern.incremental_depths;
    if (!depths || depths.length === 0) return;

    const maxDepth = Math.max(...depths) * 1.1;
    const barWidth = plotW / depths.length;

    // Draw bars
    for (let i = 0; i < depths.length; i++) {
      const barH = (depths[i] / maxDepth) * plotH;
      const x = padding.left + i * barWidth;
      const y = padding.top + plotH - barH;
      
      // Gradient color based on intensity
      const intensity = depths[i] / maxDepth;
      const r = Math.round(33 + intensity * 200);
      const g = Math.round(150 - intensity * 50);
      const b = Math.round(243 - intensity * 100);
      
      ctx.fillStyle = `rgba(${r}, ${g}, ${b}, 0.8)`;
      ctx.fillRect(x + 1, y, barWidth - 2, barH);
      
      // Highlight peak
      if (i === pattern.peak_step) {
        ctx.strokeStyle = '#ff5252';
        ctx.lineWidth = 2;
        ctx.strokeRect(x, y - 1, barWidth, barH + 1);
      }
    }

    // Draw cumulative curve
    ctx.strokeStyle = 'rgba(255, 167, 38, 0.8)';
    ctx.lineWidth = 2;
    ctx.beginPath();
    
    const fractions = pattern.cumulative_fractions;
    for (let i = 0; i < fractions.length; i++) {
      const x = padding.left + (i + 0.5) * barWidth;
      const y = padding.top + plotH - (fractions[i] * plotH);
      
      if (i === 0) ctx.moveTo(x, y);
      else ctx.lineTo(x, y);
    }
    ctx.stroke();

    // Axes
    ctx.strokeStyle = 'rgba(255,255,255,0.2)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.moveTo(padding.left, padding.top);
    ctx.lineTo(padding.left, padding.top + plotH);
    ctx.lineTo(padding.left + plotW, padding.top + plotH);
    ctx.stroke();

    ctx.font = '10px -apple-system, sans-serif';
```
