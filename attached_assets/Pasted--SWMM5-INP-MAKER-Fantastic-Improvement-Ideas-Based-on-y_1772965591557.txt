  // Fetch OSM streets client-side (no backend needed)
  async function fetchStreetNetwork(bbox) {
    const query = `
      [out:json][timeout:30];
      way["highway"~"primary|secondary|tertiary|residential"]
        (${bbox.south},${bbox.west},${bbox.north},${bbox.east});
      out geom;`;
    
    const resp = await fetch('https://overpass-api.de/api/interpreter',
      { method: 'POST', body: query });
    const data = await resp.json();
    
    // Convert OSM ways to node positions
    const nodes = extractIntersections(data);  // street intersections → junctions
    const edges = extractSegments(data);       // street segments → potential conduits
    
    return { nodes, edges };
  }
  
  // Then use existing engine with these node positions
  // instead of force-directed placement
  function generateFromStreets(streetNetwork, config) {
    // Skip force simulation — nodes already placed
    const nodePositions = streetNetwork.nodes;
    
    // Use street segments as conduit paths
    const conduits = streetNetwork.edges.map(edge => ({
      from: edge.startNode,
      to: edge.endNode,
      length: edge.length,
      // Apply INP MAKER's empirical parameters:
      roughness: sampleFromDistribution(manningDistribution),
      diameter: selectPipeSize(upstreamCount, config),
      offset: sampleOffsetPattern(offsetDistribution),
    }));
    
    // Then write all 56 sections using existing writeInp()
    return writeInp(nodePositions, conduits, config);
  }

WHY THIS IS FANTASTIC:
