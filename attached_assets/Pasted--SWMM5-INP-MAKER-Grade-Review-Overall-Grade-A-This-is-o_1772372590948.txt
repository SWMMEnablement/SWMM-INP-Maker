// ============================================
// SWMM5 INP VALIDATION ENGINE
// File: inp-validator.js
// Embeds WASM SWMM5 for instant validation
// ============================================

class InpValidator {
  constructor(options = {}) {
    this.wasmReady = false;
    this.swmmModule = null;
    this.validationResults = null;
    
    this.options = {
      wasmUrl: options.wasmUrl || '/swmm5.wasm',
      jsUrl: options.jsUrl || '/swmm5.js',
      autoFix: options.autoFix !== false, // default: auto-fix enabled
      maxRunTime: options.maxRunTime || 30, // seconds
      ...options
    };

    // Known error patterns and auto-fixes
    this.autoFixRules = [
      {
        pattern: /ERROR 138: (.+) has initial depth > max depth/,
        fix: (match, inpLines) => this.fixInitialDepth(match[1], inpLines),
        description: 'Set initial depth to 0'
      },
      {
        pattern: /ERROR 141: (.+) divides by zero/,
        fix: (match, inpLines) => this.fixDivideByZero(match[1], inpLines),
        description: 'Fix zero-value parameter'
      },
      {
        pattern: /ERROR 171: (.+) has negative slope/,
        fix: (match, inpLines) => this.fixNegativeSlope(match[1], inpLines),
        description: 'Swap invert elevations to create positive slope'
      },
      {
        pattern: /ERROR 181: (.+) has invalid cross-section/,
        fix: (match, inpLines) => this.fixCrossSection(match[1], inpLines),
        description: 'Set default circular cross-section'
      },
      {
        pattern: /ERROR 191: (.+) references undefined/,
        fix: (match, inpLines) => this.fixUndefinedReference(match[1], inpLines),
        description: 'Remove or create missing reference'
      },
      {
        pattern: /WARNING 04: (.+) has depth of zero/,
        fix: (match, inpLines) => this.fixZeroDepth(match[1], inpLines),
        description: 'Set default max depth'
      }
    ];
  }

  // ---- WASM ENGINE LOADING ----

  async loadEngine() {
    if (this.wasmReady) return;
    
    return new Promise((resolve, reject) => {
      const script = document.createElement('script');
      script.src = this.options.jsUrl;
      script.onload = () => {
        // Initialize WASM module
        if (typeof Module !== 'undefined') {
          Module.onRuntimeInitialized = () => {
            this.swmmModule = Module;
            this.wasmReady = true;
            console.log('SWMM5 WASM engine loaded for validation');
            resolve();
          };
        } else {
          reject(new Error('SWMM5 WASM module not found'));
        }
      };
      script.onerror = () => reject(new Error('Failed to load SWMM5 WASM'));
      document.head.appendChild(script);
    });
  }

  // ---- MAIN VALIDATION PIPELINE ----

  async validate(inpContent, options = {}) {
    const startTime = performance.now();
    
    const result = {
      valid: false,
      ranSuccessfully: false,
      errors: [],
      warnings: [],
      fixes: [],
      continuityError: null,
      runtime: 0,
      nodeCount: 0,
      linkCount: 0,
      subcatchCount: 0,
      timesteps: 0,
      stages: [],
      originalInp: inpContent,
      fixedInp: null,
      report: null
    };

    // Stage 1: Static analysis (no engine needed)
    result.stages.push({ name: 'Static Analysis', status: 'running' });
    const staticErrors = this.staticAnalysis(inpContent);
    result.stages[0].status = 'done';
    result.stages[0].issues = staticErrors.length;
    
    for (const err of staticErrors) {
      if (err.severity === 'error') result.errors.push(err);
      else result.warnings.push(err);
    }

    // Stage 2: Auto-fix static issues
    let workingInp = inpContent;
    if (this.options.autoFix && staticErrors.some(e => e.fixable)) {
      result.stages.push({ name: 'Auto-Repair', status: 'running' });
      const fixResult = this.autoFixStatic(workingInp, staticErrors);
      workingInp = fixResult.fixedInp;
      result.fixes.push(...fixResult.fixes);
      result.stages[1].status = 'done';
      result.stages[1].fixes = fixResult.fixes.length;
    }

    // Stage 3: Engine validation (run SWMM)
    if (this.wasmReady) {
      result.stages.push({ name: 'Engine Validation', status: 'running' });
      
      try {
        const engineResult = await this.runEngine(workingInp);
        result.ranSuccessfully = engineResult.success;
        result.report = engineResult.report;
        result.continuityError = engineResult.continuityError;
        result.runtime = engineResult.runtime;
        result.nodeCount = engineResult.nodeCount;
        result.linkCount = engineResult.linkCount;
        result.timesteps = engineResult.timesteps;
        
        // Parse engine errors/warnings
        for (const err of engineResult.errors) {
          result.errors.push({
            source: 'engine',
            message: err,
            severity: 'error',
            fixable: this.isAutoFixable(err)
          });
        }
        for (const warn of engineResult.warnings) {
          result.warnings.push({
            source: 'engine',
            message: warn,
            severity: 'warning'
          });
        }

        // Stage 4: Auto-fix engine errors
        if (this.options.autoFix && engineResult.errors.length > 0) {
          result.stages.push({ name: 'Engine Auto-Repair', status: 'running' });
          
          let attempts = 0;
          const maxAttempts = 3;
          
          while (engineResult.errors.length > 0 && attempts < maxAttempts) {
            const fixResult = this.autoFixFromErrors(workingInp, engineResult.errors);
            if (fixResult.fixes.length === 0) break;
            
            workingInp = fixResult.fixedInp;
            result.fixes.push(...fixResult.fixes);
            
            // Re-run engine
            const rerun = await this.runEngine(workingInp);
            engineResult.errors = rerun.errors;
            engineResult.warnings = rerun.warnings;
            engineResult.success = rerun.success;
            result.report = rerun.report;
            result.continuityError = rerun.continuityError;
            
            attempts++;
          }
          
          result.ranSuccessfully = engineResult.success;
          result.stages[result.stages.length - 1].status = 'done';
          result.stages[result.stages.length - 1].attempts = attempts;
        }
        
        result.stages[2].status = 'done';
      } catch (e) {
        result.stages[result.stages.length - 1].status = 'error';
        result.errors.push({
          source: 'engine',
          message: `Engine execution failed: ${e.message}`,
          severity: 'error',
          fixable: false
        });
      }
    } else {
      result.stages.push({ 
        name: 'Engine Validation', 
        status: 'skipped',
        reason: 'WASM engine not loaded'
      });
    }

    // Final verdict
    result.valid = result.errors.filter(e => e.source === 'engine').length === 0;
    result.fixedInp = workingInp !== inpContent ? workingInp : null;
    result.totalTime = performance.now() - startTime;
    
    this.validationResults = result;
    return result;
  }

  // ---- STATIC ANALYSIS (No Engine) ----

  staticAnalysis(inpContent) {
    const issues = [];
    const lines = inpContent.split('\n');
    let currentSection = null;
    
    const definedNodes = new Set();
    const definedLinks = new Set();
    const definedCurves = new Set();
    const definedTimeseries = new Set();
    const definedPatterns = new Set();
    const referencedNodes = new Set();
    const referencedCurves = new Set();
    const referencedTimeseries = new Set();
    const referencedPatterns = new Set();
    
    const nodeInverts = {};
    const nodeMaxDepths = {};
    const conduitData = {};
    
    let hasOutfall = false;
    let hasOptions = false;
    let lineNumber = 0;

    for (const line of lines) {
      lineNumber++;
      const trimmed = line.trim();
      
      if (trimmed.startsWith('[') && trimmed.endsWith(']')) {
        currentSection = trimmed.toUpperCase();
        if (currentSection === '[OPTIONS]') hasOptions = true;
        continue;
      }
      
      if (!trimmed || trimmed.startsWith(';')) continue;
      const parts = trimmed.split(/\s+/);

      switch (currentSection) {
        case '[JUNCTIONS]':
          if (parts.length >= 2) {
            definedNodes.add(parts[0]);
            nodeInverts[parts[0]] = parseFloat(parts[1]);
            nodeMaxDepths[parts[0]] = parseFloat(parts[2]) || 0;
            
            if (parseFloat(parts[1]) < -1000 || parseFloat(parts[1]) > 50000) {
              issues.push({
                source: 'static',
                severity: 'warning',
                message: `Junction ${parts[0]} has unusual invert elevation: ${parts[1]}`,
                line: lineNumber,
                fixable: false
              });
            }
            
            if (parts[2] && parseFloat(parts[2]) <= 0) {
              issues.push({
                source: 'static',
                severity: 'error',
                message: `Junction ${parts[0]} has zero or negative max depth`,
                line: lineNumber,
                fixable: true,
                fixAction: 'setDefaultDepth',
                element: parts[0]
              });
            }
          }
          break;

        case '[OUTFALLS]':
          if (parts.length >= 2) {
            definedNodes.add(parts[0]);
            hasOutfall = true;
            nodeInverts[parts[0]] = parseFloat(parts[1]);
          }
          break;

        case '[STORAGE]':
          if (parts.length >= 2) {
            definedNodes.add(parts[0]);
            nodeInverts[parts[0]] = parseFloat(parts[1]);
          }
          break;

        case '[CONDUITS]':
          if (parts.length >= 4) {
            definedLinks.add(parts[0]);
            referencedNodes.add(parts[1]);
            referencedNodes.add(parts[2]);
            
            conduitData[parts[0]] = {
              from: parts[1],
              to: parts[2],
              length: parseFloat(parts[3]),
              roughness: parseFloat(parts[4]) || 0.01
            };
            
            // Check for zero length
            if (parseFloat(parts[3]) <= 0) {
              issues.push({
                source: 'static',
                severity: 'error',
                message: `Conduit ${parts[0]} has zero or negative length`,
                line: lineNumber,
                fixable: true,
                fixAction: 'setMinLength',
                element: parts[0]
              });
            }
            
            // Check Manning's n
            const roughness = parseFloat(parts[4]);
            if (roughness && (roughness < 0.001 || roughness > 0.5)) {
              issues.push({
                source: 'static',
                severity: 'warning',
                message: `Conduit ${parts[0]} has unusual Manning's n: ${roughness}`,
                line: lineNumber,
                fixable: true,
                fixAction: 'setDefaultRoughness',
                element: parts[0]
              });
            }
          }
          break;

        case '[PUMPS]':
          if (parts.length >= 3) {
            definedLinks.add(parts[0]);
            referencedNodes.add(parts[1]);
            referencedNodes.add(parts[2]);
            if (parts[3]) referencedCurves.add(parts[3]);
          }
          break;

        case '[WEIRS]':
        case '[ORIFICES]':
          if (parts.length >= 3) {
            definedLinks.add(parts[0]);
            referencedNodes.add(parts[1]);
            referencedNodes.add(parts[2]);
          }
          break;

        case '[CURVES]':
          if (parts.length >= 1) definedCurves.add(parts[0]);
          break;

        case '[TIMESERIES]':
          if (parts.length >= 1) definedTimeseries.add(parts[0]);
          break;

        case '[PATTERNS]':
          if (parts.length >= 1) definedPatterns.add(parts[0]);
          break;

        case '[DWF]':
          if (parts.length >= 2) {
            referencedNodes.add(parts[0]);
            // Check for pattern references
            for (let i = 3; i < parts.length; i++) {
              if (parts[i] && !parts[i].startsWith('"') && isNaN(parseFloat(parts[i]))) {
                referencedPatterns.add(parts[i]);
              }
            }
          }
          break;

        case '[RAINGAGES]':
          if (parts.length >= 4) {
            // Check for timeseries reference
            for (let i = 0; i < parts.length; i++) {
              if (parts[i] === 'TIMESERIES' && parts[i+1]) {
                referencedTimeseries.add(parts[i+1]);
              }
            }
          }
          break;
      }
    }

    // Cross-reference checks
    
    // Missing outfall
    if (!hasOutfall) {
      issues.push({
        source: 'static',
        severity: 'error',
        message: 'No outfall defined — model cannot run without at least one outfall',
        fixable: true,
        fixAction: 'addOutfall'
      });
    }

    // Missing OPTIONS
    if (!hasOptions) {
      issues.push({
        source: 'static',
        severity: 'error',
        message: 'No [OPTIONS] section — required for simulation',
        fixable: true,
        fixAction: 'addOptions'
      });
    }

    // Orphan nodes (defined but no links connect to them)
    for (const node of definedNodes) {
      if (!referencedNodes.has(node) && !this.isOutfall(node, lines)) {
        issues.push({
          source: 'static',
          severity: 'warning',
          message: `Node ${node} is disconnected — no links reference it`,
          fixable: false
        });
      }
    }

    // Undefined node references
    for (const node of referencedNodes) {
      if (!definedNodes.has(node)) {
        issues.push({
          source: 'static',
          severity: 'error',
          message: `Link references undefined node: ${node}`,
          fixable: true,
          fixAction: 'addMissingNode',
          element: node
        });
      }
    }

    // Undefined curve references
    for (const curve of referencedCurves) {
      if (!definedCurves.has(curve)) {
        issues.push({
          source: 'static',
          severity: 'error',
          message: `Element references undefined curve: ${curve}`,
          fixable: true,
          fixAction: 'addDefaultCurve',
          element: curve
        });
      }
    }

    // Adverse slopes (invert check)
    for (const [id, data] of Object.entries(conduitData)) {
      const fromInvert = nodeInverts[data.from];
      const toInvert = nodeInverts[data.to];
      
      if (fromInvert !== undefined && toInvert !== undefined) {
        if (toInvert > fromInvert) {
          issues.push({
            source: 'static',
            severity: 'warning',
            message: `Conduit ${id}: adverse slope — downstream invert (${toInvert.toFixed(2)}) > upstream (${fromInvert.toFixed(2)})`,
            fixable: true,
            fixAction: 'fixAdverseSlope',
            element: id,
            data: { from: data.from, to: data.to, fromInvert, toInvert }
          });
        }
        
        // Check for very flat slope
        if (data.length > 0) {
          const slope = Math.abs(fromInvert - toInvert) / data.length;
          if (slope < 0.0001 && slope > 0) {
            issues.push({
              source: 'static',
              severity: 'warning',
              message: `Conduit ${id}: very flat slope (${(slope * 100).toFixed(4)}%) — may cause instability`,
              fixable: false
            });
          }
        }
      }
    }

    return issues;
  }

  isOutfall(nodeId, lines) {
    let inOutfalls = false;
    for (const line of lines) {
      if (line.trim() === '[OUTFALLS]') { inOutfalls = true; continue; }
      if (line.trim().startsWith('[')) { inOutfalls = false; continue; }
      if (inOutfalls && line.trim().startsWith(nodeId)) return true;
    }
    return false;
  }

  // ---- AUTO-FIX ENGINE ----

  autoFixStatic(inpContent, issues) {
    let fixedInp = inpContent;
    const fixes = [];
    const fixableIssues = issues.filter(i => i.fixable);

    for (const issue of fixableIssues) {
      let fixed = false;
      
      switch (issue.fixAction) {
        case 'setDefaultDepth':
          fixedInp = this.replaceNodeDepth(fixedInp, issue.element, 4.0);
          fixed = true;
          break;

        case 'setMinLength':
          fixedInp = this.replaceConduitLength(fixedInp, issue.element, 10.0);
          fixed = true;
          break;

        case 'setDefaultRoughness':
          fixedInp = this.replaceConduitRoughness(fixedInp, issue.element, 0.013);
          fixed = true;
          break;

        case 'fixAdverseSlope':
          if (issue.data) {
            // Lower the downstream invert to create 0.1% minimum slope
            const length = this.getConduitLength(fixedInp, issue.element);
            const newToInvert = issue.data.fromInvert - (length * 0.001);
            fixedInp = this.replaceNodeInvert(fixedInp, issue.data.to, newToInvert);
            fixed = true;
          }
          break;

        case 'addOutfall':
          fixedInp = this.addDefaultOutfall(fixedInp);
          fixed = true;
          break;

        case 'addOptions':
          fixedInp = this.addDefaultOptions(fixedInp);
          fixed = true;
          break;

        case 'addMissingNode':
          fixedInp = this.addDefaultJunction(fixedInp, issue.element);
          fixed = true;
          break;

        case 'addDefaultCurve':
          fixedInp = this.addDefaultPumpCurve(fixedInp, issue.element);
          fixed = true;
          break;
      }

      if (fixed) {
        fixes.push({
          issue: issue.message,
          action: issue.fixAction,
          element: issue.element,
          description: `Auto-fixed: ${issue.message}`
        });
      }
    }

    return { fixedInp, fixes };
  }

  autoFixFromErrors(inpContent, errors) {
    let fixedInp = inpContent;
    const fixes = [];

    for (const error of errors) {
      for (const rule of this.autoFixRules) {
        const match = error.match(rule.pattern);
        if (match) {
          try {
            fixedInp = rule.fix(match, fixedInp);
            fixes.push({
              issue: error,
              action: rule.description,
              element: match[1]
            });
          } catch (e) {
            console.warn(`Auto-fix failed for: ${error}`, e);
          }
          break;
        }
      }
    }

    return { fixedInp, fixes };
  }

  isAutoFixable(errorMessage) {
    return this.autoFixRules.some(rule => rule.pattern.test(errorMessage));
  }

  // ---- FIX HELPERS ----

  replaceNodeDepth(inp, nodeId, newDepth) {
    const lines = inp.split('\n');
    let inJunctions = false;
    
    for (let i = 0; i < lines.length; i++) {
      if (lines[i].trim() === '[JUNCTIONS]') { inJunctions = true; continue; }
      if (lines[i].trim().startsWith('[')) { inJunctions = false; continue; }
      
      if (inJunctions) {
        const parts = lines[i].trim().split(/\s+/);
        if (parts[0] === nodeId && parts.length >= 2) {
          parts[2] = newDepth.toFixed(2);
          lines[i] = parts.join('\t');
        }
      }
    }
    
    return lines.join('\n');
  }

  replaceNodeInvert(inp, nodeId, newInvert) {
    const lines = inp.split('\n');
    const sections = ['[JUNCTIONS]', '[OUTFALLS]', '[STORAGE]'];
    let currentSection = null;
    
    for (let i = 0; i < lines.length; i++) {
      const trimmed = lines[i].trim();
      if (trimmed.startsWith('[')) { 
        currentSection = trimmed.toUpperCase();
        continue;
      }
      
      if (sections.includes(currentSection)) {
        const parts = trimmed.split(/\s+/);
        if (parts[0] === nodeId && parts.length >= 2) {
          parts[1] = newInvert.toFixed(2);
          lines[i] = parts.join('\t');
        }
      }
    }

    const lines = inp.split('\n');
    let inSection = false;
    
    for (let i = 0; i < lines.length; i++) {
      if (lines[i].trim() === section) { inSection = true; continue; }
      if (lines[i].trim().startsWith('[')) { inSection = false; continue; }
      
      if (inSection) {
        const parts = lines[i].trim().split(/\s+/);
        if (parts[0] === elementId) {
          if (colIndex >= 0 && colIndex < parts.length) {
            parts[colIndex] = newValue;
            lines[i] = parts.join('\t');
          } else if (colIndex === -1) {
            lines[i] = newValue;

      }
    }

  async runEngine(inpContent) {
    if (!this.wasmReady) {
      throw new Error('WASM engine not loaded');
    }

    return new Promise((resolve) => {
      try {
        // Write .inp to virtual filesystem
        const encoder = new TextEncoder();
        const inpBytes = encoder.encode(inpContent);
        this.swmmModule.FS.writeFile('/model.inp', inpBytes);

        // Run SWMM
        const startTime = performance.now();
        const returnCode = this.swmmModule._swmm_run(
          this.swmmModule.allocateUTF8('/model.inp'),
          this.swmmModule.allocateUTF8('/model.rpt'),
          this.swmmModule.allocateUTF8('/model.out')
        );
        const runtime = performance.now() - startTime;

        // Read report
        let report = '';
        try {
          const rptBytes = this.swmmModule.FS.readFile('/model.rpt');
          report = new TextDecoder().decode(rptBytes);
        } catch (e) {
          report = '';
        }

        // Parse results
        const errors = [];
        const warnings = [];
        let continuityError = null;
        let nodeCount = 0;
        let linkCount = 0;
        let timesteps = 0;

        for (const line of report.split('\n')) {
          const trimmed = line.trim();
          
          if (trimmed.startsWith('ERROR')) {
            errors.push(trimmed);
          }
          if (trimmed.startsWith('WARNING')) {
            warnings.push(trimmed);
          }
          if (trimmed.includes('Runoff Quantity Continuity') || 
              trimmed.includes('Flow Routing Continuity')) {
            // Look for error percentage
            const pctMatch = trimmed.match(/([\d.]+)\s*%/);
            if (pctMatch) {
              const pct = parseFloat(pctMatch[1]);
              if (continuityError === null || pct > continuityError) {
                continuityError = pct;
              }
            }
          }
        }

        // Cleanup virtual files
        try {
          this.swmmModule.FS.unlink('/model.inp');
          this.swmmModule.FS.unlink('/model.rpt');
          this.swmmModule.FS.unlink('/model.out');
        } catch (e) { /* ignore cleanup errors */ }

        resolve({
          success: returnCode === 0 && errors.length === 0,
          returnCode,
          errors,
          warnings,
          report,
          continuityError,
          nodeCount,
          linkCount,
          timesteps,
          runtime
        });

      } catch (e) {
        resolve({
          success: false,
          returnCode: -1,
          errors: [e.message],
          warnings: [],
          report: '',
          continuityError: null,
          runtime: 0
        });
      }
    });
  }
}
